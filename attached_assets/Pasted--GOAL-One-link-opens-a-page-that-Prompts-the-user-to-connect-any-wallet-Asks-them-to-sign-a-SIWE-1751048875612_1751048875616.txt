// App.jsx
import { useAccount, useConnect, useDisconnect, useSignMessage, useEnsAddress, useEnsName } from 'wagmi';
import { getDefaultConfig, RainbowKitProvider, ConnectButton } from '@rainbow-me/rainbowkit';
import { WagmiProvider, createConfig } from 'wagmi';
import { mainnet } from 'wagmi/chains';
import { SiweMessage } from 'siwe';
import { useState } from 'react';

const config = getDefaultConfig({
  appName: 'ENS Proof',
  projectId: 'YOUR_WALLETCONNECT_PROJECT_ID', // get from https://cloud.walletconnect.com/
  chains: [mainnet],
});

function ProveENS() {
  const [result, setResult] = useState('');
  const { address, isConnected } = useAccount();
  const { data: ensName } = useEnsName({ address });
  const { data: ensAddress } = useEnsAddress({ name: 'hansen.ipecity.eth' });
  const { signMessageAsync } = useSignMessage();

  const prove = async () => {
    if (!isConnected || !ensAddress) return;
    if (address.toLowerCase() !== ensAddress.toLowerCase()) {
      setResult(`❌ Not owner of hansen.ipecity.eth`);
      return;
    }

    const msg = new SiweMessage({
      domain: window.location.host,
      address,
      statement: 'I prove I own hansen.ipecity.eth',
      uri: window.location.origin,
      version: '1',
      chainId: 1,
      nonce: crypto.randomUUID(),
      issuedAt: new Date().toISOString(),
    }).prepareMessage();

    const signature = await signMessageAsync({ message: msg });
    const recovered = await import('viem').then(({ recoverMessageAddress }) =>
      recoverMessageAddress({ message: msg, signature })
    );

    if (recovered.toLowerCase() === address.toLowerCase()) {
      setResult(`✅ Verified!\n\n${msg}\n\nSignature:\n${signature}`);
    } else {
      setResult(`❌ Signature invalid`);
    }
  };

  return (
    <div>
      <ConnectButton />
      <button onClick={prove} disabled={!isConnected}>Prove ENS</button>
      <pre>{result}</pre>
    </div>
  );
}

export default function App() {
  return (
    <WagmiProvider config={config}>
      <RainbowKitProvider>
        <ProveENS />
      </RainbowKitProvider>
    </WagmiProvider>
  );
}
